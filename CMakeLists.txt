cmake_minimum_required(VERSION 3.20)
project(Luna LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/l*.c")
list(REMOVE_ITEM LUA_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/lua.c" "${CMAKE_CURRENT_SOURCE_DIR}/ltests.c")
add_library(luna_runtime STATIC ${LUA_SOURCES})
target_include_directories(luna_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(lua lua.c)
target_link_libraries(lua PRIVATE luna_runtime)
if(MSVC)
  target_compile_definitions(luna_runtime PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_definitions(lua PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_link_libraries(luna_runtime PUBLIC m)
endif()
